Q2Q2Merge Sort on strings, counting inversion swaps

Pseudocode 
function mergeSort(s, lo, hi):
  if lo >= hi: return 0
  mid = (lo + hi) / 2
  inv  = mergeSort(s, lo, mid)
  inv += mergeSort(s, mid+1, hi)
  inv += merge(s, lo, mid, hi)
  return inv

function merge(s, lo, mid, hi):
  i = lo; j = mid+1; inv = 0; tmp = []
  while i <= mid and j <= hi:
    if strcmp(s[i], s[j]) <= 0: tmp.add(s[i]); i++
    else: tmp.add(s[j]); j++; inv += (mid - i + 1)
  copy the rest of both halves into tmp
  copy tmp back into s[lo..hi]
  return inv
C code: q02_merge_sort_strings.c 
#include <stdio.h>
#include <string.h>

char s[100][50];
char t[100][50];
long long inv = 0;

void merge(int lo, int mid, int hi) {
    int i = lo, j = mid + 1, k = lo;
    while (i <= mid && j <= hi) {
        if (strcmp(s[i], s[j]) <= 0) {
            strcpy(t[k++], s[i++]);
        } else {
            strcpy(t[k++], s[j++]);
            inv += mid - i + 1;   /* right string jumps over the rest of left half */
        }
    }
    while (i <= mid) strcpy(t[k++], s[i++]);
    while (j <= hi)  strcpy(t[k++], s[j++]);
    for (k = lo; k <= hi; k++)
        strcpy(s[k], t[k]);
}

void mergeSort(int lo, int hi) {
    if (lo >= hi) return;
    int mid = (lo + hi) / 2;
    mergeSort(lo, mid);
    mergeSort(mid + 1, hi);
    merge(lo, mid, hi);
}

int main() {
    int n, i;
    printf("How many strings: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        printf("String %d: ", i + 1);
        scanf("%49s", s[i]);
    }

    mergeSort(0, n - 1);

    printf("Sorted:\n");
    for (i = 0; i < n; i++)
        printf("%s\n", s[i]);
    printf("Inversion swaps: %lld\n", inv);
    return 0;
}

Q3Q3Quick Sort with Lomuto partition and random pivot

Pseudocode 
function quick(a, low, high):
  if low < high:
    r = random(low, high)
    swap a[r], a[high]
    p = partition(a, low, high)
    quick(a, low, p-1)
    quick(a, p+1, high)

function partition(a, low, high):
  pivot = a[high]; i = low - 1
  for j = low to high-1:
    if a[j] <= pivot: i++; swap a[i], a[j]
  swap a[i+1], a[high]
  return i+1
C code: q03_quick_sort.c 
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

void swap(int *x, int *y) {
    int t = *x; *x = *y; *y = t;
}

int partition(int a[], int lo, int hi) {
    int pivot = a[hi];
    int i = lo - 1, j;
    for (j = lo; j < hi; j++) {
        if (a[j] <= pivot) {
            i++;
            swap(&a[i], &a[j]);
        }
    }
    swap(&a[i + 1], &a[hi]);
    return i + 1;
}

void quickSort(int a[], int lo, int hi) {
    if (lo < hi) {
        int r = lo + rand() % (hi - lo + 1);   /* random pivot */
        swap(&a[r], &a[hi]);
        int p = partition(a, lo, hi);
        quickSort(a, lo, p - 1);
        quickSort(a, p + 1, hi);
    }
}

int main() {
    int n, i, a[100];
    srand(time(NULL));
    printf("How many numbers: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        printf("Enter number %d: ", i + 1);
        scanf("%d", &a[i]);
    }

    quickSort(a, 0, n - 1);

    printf("Sorted: ");
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\n");
    return 0;
}

Q5Q5Linked Stack ADT with exception handling

Pseudocode 
push(s, x):
  node = new Node(x, next = s.head); s.head = node

isEmpty(s): return s.head == NULL

pop(s):
  if isEmpty(s): error "stack underflow"
  x = s.head.value; s.head = s.head.next; return x

  if isEmpty(s): error "stack empty"
  return s.head.value
C code: q05_linked_stack.c 
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int value;
    struct Node *next;
};

struct Node *top = NULL;

int isEmpty() {
    return top == NULL;
}

void push(int x) {
    struct Node *n = malloc(sizeof(struct Node));
    n->value = x;
    n->next = top;
    top = n;
}

int pop(int *x) {
    if (isEmpty()) return 0;
    struct Node *t = top;
    *x = t->value;
    top = t->next;
    free(t);
    return 1;
}

int peek(int *x) {
    if (isEmpty()) return 0;
    *x = top->value;
    return 1;
}

int main() {
    int ch, x;
    while (1) {
        printf("\n1 Push  2 Pop  3 Peek  4 Is empty  0 Exit\nChoice: ");
        scanf("%d", &ch);
        if (ch == 0) break;

        if (ch == 1) {
            printf("Value: ");
            scanf("%d", &x);
            push(x);
            printf("Pushed.\n");
        }
        else if (ch == 2) {
            if (pop(&x)) printf("Popped: %d\n", x);
            else printf("Error: stack underflow\n");
        }
        else if (ch == 3) {
            if (peek(&x)) printf("Top: %d\n", x);
            else printf("Error: stack is empty\n");
        }
        else if (ch == 4) {
            printf(isEmpty() ? "Yes, empty\n" : "No, not empty\n");
        }
    }
    return 0;
}

C code: q07_doubly_linked_list.c 
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int value;
    struct Node *prev, *next;
};

struct Node *head = NULL, *tail = NULL;

void insertFront(int x) {
    struct Node *n = malloc(sizeof(struct Node));
    n->value = x;
    n->prev = NULL;
    n->next = head;
    if (head) head->prev = n;
    else tail = n;
    head = n;
}

void insertEnd(int x) {
    struct Node *n = malloc(sizeof(struct Node));
    n->value = x;
    n->next = NULL;
    n->prev = tail;
    if (tail) tail->next = n;
    else head = n;
    tail = n;
}

int deleteValue(int x) {
    struct Node *p = head;
    while (p && p->value != x) p = p->next;
    if (!p) return 0;

    if (p->prev) p->prev->next = p->next;
    else head = p->next;

    if (p->next) p->next->prev = p->prev;
    else tail = p->prev;

    free(p);
    return 1;
}

int main() {
    int ch, x;
    struct Node *p;
    while (1) {
        printf("\n1 Insert front  2 Insert end  3 Delete value\n");
        printf("4 Show forward   5 Show reverse   0 Exit\nChoice: ");
        scanf("%d", &ch);
        if (ch == 0) break;

        if (ch == 1) {
            printf("Value: ");
            scanf("%d", &x);
            insertFront(x);
        }
        else if (ch == 2) {
            printf("Value: ");
            scanf("%d", &x);
            insertEnd(x);
        }
        else if (ch == 3) {
            printf("Value to delete: ");
            scanf("%d", &x);
            if (!deleteValue(x)) printf("Not found\n");
        }
        else if (ch == 4) {
            for (p = head; p; p = p->next) printf("%d ", p->value);
            printf("\n");
        }
        else if (ch == 5) {
            for (p = tail; p; p = p->prev) printf("%d ", p->value);
            printf("\n");
        }
    }
    return 0;
}

Q8Q8Circular Linked List as a round-robin scheduler

Pseudocode 
quantum = 3
cur = head  // circular: tail.next == head

while list not empty:
  t = min(quantum, cur.remaining)
  print cur.name, "ran for", t
  cur.remaining -= t; clock += t
  if cur.remaining == 0: remove cur from ring
  else: cur = cur.next
C code: q08_round_robin.c 
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int id, remaining;
    struct Node *next;
};

int main() {
    int n, q, i, clock = 0;
    struct Node *head = NULL, *tail = NULL, *cur, *p;

    printf("Number of processes: ");
    scanf("%d", &n);
    printf("Time quantum: ");
    scanf("%d", &q);

    for (i = 0; i < n; i++) {
        p = malloc(sizeof(struct Node));
        p->id = i + 1;
        printf("Burst time of P%d: ", i + 1);
        scanf("%d", &p->remaining);
        if (!head) {
            head = tail = p;
        } else {
            tail->next = p;
            tail = p;
        }
        p->next = head;          /* keep the ring closed */
    }

    cur = head;
    while (head) {
        int t = cur->remaining < q ? cur->remaining : q;
        clock += t;
        cur->remaining -= t;
        printf("P%d ran for %d (time now %d)\n", cur->id, t, clock);

        if (cur->remaining == 0) {
            if (cur->next == cur) {          /* last process left */
                free(cur);
                head = NULL;
                break;
            }
            p = head;
            while (p->next != cur) p = p->next;   /* find the node before cur */
            p->next = cur->next;
            if (head == cur) head = cur->next;
            if (tail == cur) tail = p;
            struct Node *done = cur;
            cur = cur->next;
            free(done);
        } else {
            cur = cur->next;
        }
    }
    printf("All processes finished.\n");
    return 0;
}

Q9Q9Infix to Postfix conversion

Pseudocode 
for each token t in expression:
  if t is operand: output t
  elif t == "(": push t
  elif t == ")":
    while top != "(": output pop
    pop  // remove "("
  else:  // operator
    while stack not empty and top != "(" and
          (prec(top) > prec(t) or
           (prec(top) == prec(t) and t is not "^")):
      output pop
    push t
while stack not empty: output pop
C code: q09_infix_to_postfix.c 
#include <stdio.h>

char st[100];
int top = -1;

int prec(char c) {
    if (c == '^') return 3;
    if (c == '*' || c == '/') return 2;
    if (c == '+' || c == '-') return 1;
    return 0;
}

int main() {
    char e[100], out[100];
    int i, k = 0;

    printf("Enter infix expression (no spaces, single-letter or digit operands): ");
    scanf("%99s", e);

    for (i = 0; e[i]; i++) {
        char c = e[i];
        if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9')) {
            out[k++] = c;
        }
        else if (c == '(') {
            st[++top] = c;
        }
        else if (c == ')') {
            while (top >= 0 && st[top] != '(') out[k++] = st[top--];
            top--;   /* remove '(' */
        }
        else {   /* operator */
            while (top >= 0 && st[top] != '(' &&
                   (prec(st[top]) > prec(c) ||
                    (prec(st[top]) == prec(c) && c != '^'))) {
                out[k++] = st[top--];
            }
            st[++top] = c;
        }
    }
    while (top >= 0) out[k++] = st[top--];
    out[k] = '\0';

    printf("Postfix: %s\n", out);
    return 0;
}

    char s[100];
    int i;

    printf("Enter expression (no spaces): ");
    scanf("%99s", s);

    for (i = 0; s[i]; i++) {
        char c = s[i];
        if (c == '(' || c == '[' || c == '{') {
            st[++top] = c;
        }
        else if (c == ')' || c == ']' || c == '}') {
            if (top < 0) {
                printf("Not balanced\n");
                return 0;

                (c == ']' && open != '[') ||

                return 0;
            }
        }
    }

}

C code: q13_linked_tree_traversals.c 
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int value;
    struct Node *left, *right;
};

/* Values are inserted in BST order, so the tree is linked and easy to build. */
struct Node *insert(struct Node *root, int x) {
    if (!root) {
        root = malloc(sizeof(struct Node));
        root->value = x;
        root->left = root->right = NULL;
        return root;
    }
    if (x < root->value) root->left = insert(root->left, x);
    else                 root->right = insert(root->right, x);
    return root;
}

/* Iterative inorder: uses an explicit stack instead of recursion */
void inorderIterative(struct Node *root) {
    struct Node *stack[100];
    int top = -1;
    struct Node *cur = root;

    while (cur || top >= 0) {
        while (cur) {
            stack[++top] = cur;
            cur = cur->left;
        }
        cur = stack[top--];
        printf("%d ", cur->value);
        cur = cur->right;
    }
    printf("\n");
}

/* Level order: uses an explicit queue */
void levelOrder(struct Node *root) {
    if (!root) return;
    struct Node *queue[100];
    int f = 0, r = 0;
    queue[r++] = root;

    while (f < r) {
        struct Node *n = queue[f++];
        printf("%d ", n->value);
        if (n->left)  queue[r++] = n->left;
        if (n->right) queue[r++] = n->right;
    }
    printf("\n");
}

int main() {
    struct Node *root = NULL;
    int n, i, x;

    printf("How many values: ");
    scanf("%d", &n);
    for (i = 0; i < n; i++) {
        printf("Value %d: ", i + 1);
        scanf("%d", &x);
        root = insert(root, x);
    }

    printf("Inorder (iterative): ");
    inorderIterative(root);
    printf("Level order:         ");
    levelOrder(root);
    return 0;
}

unsigned long long merkleRoot(char tx[][100], int n) {

        lv[i] = h(tx[i]);

}

struct Node {
    int key;
    struct Node *left, *right;
};

struct Node *rotateRight(struct Node *y) {
    struct Node *x = y->left;
    y->left = x->right;
    x->right = y;
    return x;
}

struct Node *rotateLeft(struct Node *x) {
    struct Node *y = x->right;
    x->right = y->left;
    y->left = x;
    return y;
}

/* Move the node with key k (or the last node visited) to the root */
struct Node *splay(struct Node *root, int k) {
    if (!root || root->key == k) return root;

    if (k < root->key) {
        if (!root->left) return root;
        if (k < root->left->key) {                    /* zig-zig */
            root->left->left = splay(root->left->left, k);
            root = rotateRight(root);
        } else if (k > root->left->key) {             /* zig-zag */
            root->left->right = splay(root->left->right, k);
            if (root->left->right) root->left = rotateLeft(root->left);
        }
        return root->left ? rotateRight(root) : root;
    } else {
        if (!root->right) return root;
        if (k > root->right->key) {                   /* zig-zig */
            root->right->right = splay(root->right->right, k);
            root = rotateLeft(root);
        } else if (k < root->right->key) {            /* zig-zag */
            root->right->left = splay(root->right->left, k);
            if (root->right->left) root->right = rotateRight(root->right);
        }
        return root->right ? rotateLeft(root) : root;
    }
}

    if (!root) {
        struct Node *n = malloc(sizeof(struct Node));
        n->key = k;
        n->left = n->right = NULL;
        return n;
    }
    root = splay(root, k);
    if (root->key == k) return root;   /* already present, now at root */

    struct Node *n = malloc(sizeof(struct Node));
    n->key = k;
    if (k < root->key) {
        n->left = root->left;
        n->right = root;
        root->left = NULL;
    } else {
        n->right = root->right;
        n->left = root;
        root->right = NULL;
    }

Q25Q25Cycle detection in a directed graph with DFS

Pseudocode 
hasCycleDFS(u):
  visited[u] = true; onStack[u] = true
  for v in adj[u]:               // directed edges u -> v
    if onStack[v]: return true   // back edge
    if not visited[v] and hasCycleDFS(v): return true
  onStack[u] = false
  return false

detectCycle():
  visited[], onStack[] = false
  for u = 0 to n-1:
    if not visited[u] and hasCycleDFS(u): return "Cycle found"
  return "No cycle"
C code: q25_directed_cycle.c 
#include <stdio.h>
#include <stdlib.h>

#define MAX 50

struct Edge {
    int to;
    struct Edge *next;
};

struct Edge *adj[MAX];
int visited[MAX];
int onStack[MAX];   /* vertices on the current DFS path */
int n;

void addEdge(int u, int v) {          /* directed: u -> v only */
    struct Edge *e = malloc(sizeof(struct Edge));
    e->to = v; e->next = adj[u]; adj[u] = e;
}

int hasCycle(int u) {
    visited[u] = 1;
    onStack[u] = 1;
    for (struct Edge *e = adj[u]; e; e = e->next) {
        if (onStack[e->to]) return 1;                 /* back edge: cycle */
        if (!visited[e->to] && hasCycle(e->to)) return 1;
    }
    onStack[u] = 0;   /* leaving this vertex */
    return 0;
}

int main() {
    int m, i, u, v;

    printf("Number of vertices (0 to n-1): ");
    scanf("%d", &n);
    printf("Number of directed edges: ");
    scanf("%d", &m);

    printf("Enter %d edges as 'from to':\n", m);
    for (i = 0; i < m; i++) {
        scanf("%d %d", &u, &v);
        addEdge(u, v);
    }

    int found = 0;
    for (u = 0; u < n && !found; u++)
        if (!visited[u] && hasCycle(u))
            found = 1;

    printf(found ? "Cycle detected\n" : "No cycle\n");
    return 0;
}
